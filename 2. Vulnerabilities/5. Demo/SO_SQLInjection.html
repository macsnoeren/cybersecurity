# VULNERABLE
def vulnerable_login(u, p):
q = f"SELECT * FROM users
WHERE username = '{u}'
AND password = '{p}'"
cursor.execute(q)
# SAFE
def safe_login(u, p):
q = "SELECT * FROM users
WHERE username = ?
AND password = ?"
cursor.execute(q, (u, p))
Kern: bij f"...'{u}'..." wordt de input ín de SQL-tekst geplakt. Een ' sluit de string vroegtijdig af, waarna OR '1'='1' als code wordt uitgevoerd. Een parameterized query stuurt de input apart mee als data (?), dus quotes hebben geen betekenis meer.
